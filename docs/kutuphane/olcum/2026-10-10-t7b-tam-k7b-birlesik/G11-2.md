Tek dosya. `index.html` olarak kaydedip tarayıcıda açman yeterli, build ya da kurulum gerekmiyor.

```html
<!doctype html>
<html>
<head>
  <meta charset="utf-8">
  <style>body{margin:0;overflow:hidden}</style>
  <script type="importmap">
    { "imports": { "three": "https://cdn.jsdelivr.net/npm/three@0.170.0/build/three.module.js" } }
  </script>
</head>
<body>
<script type="module">
import * as THREE from 'three';

const scene = new THREE.Scene();
scene.background = new THREE.Color(0x202530);
const camera = new THREE.PerspectiveCamera(60, innerWidth / innerHeight, 0.1, 100);
const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(innerWidth, innerHeight);
document.body.appendChild(renderer.domElement);

scene.add(new THREE.HemisphereLight(0xffffff, 0x444444, 2));
scene.add(new THREE.GridHelper(50, 50));

const cube = new THREE.Mesh(
  new THREE.BoxGeometry(),
  new THREE.MeshStandardMaterial({ color: 0x44aa88 })
);
cube.position.y = 0.5;
scene.add(cube);

// e.code klavye düzeninden bağımsız (Türkçe Q/F klavyede de çalışır)
const keys = {};
addEventListener('keydown', e => keys[e.code] = true);
addEventListener('keyup', e => keys[e.code] = false);

const offset = new THREE.Vector3(0, 4, 8); // küpün arkası + yukarısı
const camTarget = new THREE.Vector3();
const dir = new THREE.Vector3();
const clock = new THREE.Clock();
camera.position.copy(cube.position).add(offset);

renderer.setAnimationLoop(() => {
  const dt = clock.getDelta();

  dir.set(
    (keys.KeyD ? 1 : 0) - (keys.KeyA ? 1 : 0),
    0,
    (keys.KeyS ? 1 : 0) - (keys.KeyW ? 1 : 0)
  ).normalize(); // çapraz giderken hız artmasın
  cube.position.addScaledVector(dir, 5 * dt);

  // Kamera hedefe kare hızından bağımsız, yumuşak yaklaşır
  camTarget.copy(cube.position).add(offset);
  camera.position.lerp(camTarget, 1 - Math.exp(-5 * dt));
  camera.lookAt(cube.position);

  renderer.render(scene, camera);
});

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
</script>
</body>
</html>
```

- **Hareket:** WASD dünya eksenlerinde hareket ettirir. Küp dönmez, kamera da her zaman aynı sabit açıdan takip eder.
- **Atladıklarım:** küpün yönüne göre hareket ve dönen kamera. Gerekirse WASD küpün yönüne göre çalışsın ve kamera arkasında dönsün diye `offset` değerine `applyQuaternion(cube.quaternion)` uygulanabilir.
- **Kontrol:** kodu tarayıcıda çalıştırmadım.
- **Skill'ler:** departman-*/frontend-craft skill'lerini açmadım, çünkü bu tek dosyalık bir örnek ve projede değişiklik yok.
