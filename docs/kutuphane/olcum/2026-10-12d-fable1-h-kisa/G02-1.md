using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddDbContext<AppDbContext>(o => o.UseInMemoryDatabase("urunler"));

var app = builder.Build();

app.MapGet("/urunler/{id:int}", async (int id, AppDbContext db) =>
    await db.Urunler.FindAsync(id) is { } urun ? Results.Ok(urun) : Results.NotFound());

app.Run();

public partial class Program { }

public class Urun
{
    public int Id { get; set; }
    public string Ad { get; set; } = "";
    public decimal Fiyat { get; set; }
}

public class AppDbContext(DbContextOptions<AppDbContext> options) : DbContext(options)
{
    public DbSet<Urun> Urunler => Set<Urun>();
}

// UrunlerTests.cs
using System.Net;
using System.Net.Http.Json;
using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.Extensions.DependencyInjection;

public class UrunlerTests(WebApplicationFactory<Program> factory) : IClassFixture<WebApplicationFactory<Program>>
{
    [Fact]
    public async Task GetUrun_VarOlanId_200VeUrunDoner()
    {
        Seed(factory, new Urun { Id = 1, Ad = "Kalem", Fiyat = 12.5m });
        var client = factory.CreateClient();

        var response = await client.GetAsync("/urunler/1");

        Assert.Equal(HttpStatusCode.OK, response.StatusCode);
        var urun = await response.Content.ReadFromJsonAsync<Urun>();
        Assert.NotNull(urun);
        Assert.Equal("Kalem", urun.Ad);
    }

    [Fact]
    public async Task GetUrun_YokId_404Doner()
    {
        var client = factory.CreateClient();

        var response = await client.GetAsync("/urunler/999999");

        Assert.Equal(HttpStatusCode.NotFound, response.StatusCode);
    }

    private static void Seed(WebApplicationFactory<Program> factory, Urun urun)
    {
        using var scope = factory.Services.CreateScope();
        var db = scope.ServiceProvider.GetRequiredService<AppDbContext>();
        db.Database.EnsureCreated();
        if (!db.Urunler.Any(u => u.Id == urun.Id))
        {
            db.Urunler.Add(urun);
            db.SaveChanges();
        }
    }
}
