// UrunEndpointTests.cs
using System.Net;
using System.Net.Http.Json;
using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;
using Xunit;

public class UrunEndpointTests : IClassFixture<WebApplicationFactory<Program>>
{
    private readonly WebApplicationFactory<Program> _factory;

    public UrunEndpointTests(WebApplicationFactory<Program> factory)
    {
        _factory = factory.WithWebHostBuilder(b => b.ConfigureServices(s =>
        {
            s.RemoveAll<DbContextOptions<UrunDbContext>>();
            s.AddDbContext<UrunDbContext>(o => o.UseInMemoryDatabase("urunler-test"));
        }));
    }

    [Fact]
    public async Task GET_urun_varsa_200_ve_urun_doner()
    {
        using (var scope = _factory.Services.CreateScope())
        {
            var db = scope.ServiceProvider.GetRequiredService<UrunDbContext>();
            db.Urunler.Add(new Urun { Id = 1, Ad = "Kalem", Fiyat = 12.5m });
            await db.SaveChangesAsync();
        }

        var response = await _factory.CreateClient().GetAsync("/urunler/1");

        response.EnsureSuccessStatusCode();
        var urun = await response.Content.ReadFromJsonAsync<Urun>();
        Assert.Equal("Kalem", urun!.Ad);
    }

    [Fact]
    public async Task GET_urun_yoksa_404_doner()
    {
        var response = await _factory.CreateClient().GetAsync("/urunler/999");

        Assert.Equal(HttpStatusCode.NotFound, response.StatusCode);
    }
}
