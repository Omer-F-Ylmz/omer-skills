// Program.cs
using Microsoft.AspNetCore.Http.HttpResults;
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddDbContext<AppDb>(o => o.UseInMemoryDatabase("urunler"));
var app = builder.Build();

app.MapGet("/urunler/{id:int}", UrunEndpoints.Getir);

app.Run();

public record Urun(int Id, string Ad, decimal Fiyat);

public class AppDb(DbContextOptions<AppDb> options) : DbContext(options)
{
    public DbSet<Urun> Urunler => Set<Urun>();
}

public static class UrunEndpoints
{
    public static async Task<Results<Ok<Urun>, NotFound>> Getir(int id, AppDb db)
    {
        var urun = await db.Urunler.FindAsync(id);
        return urun is null ? TypedResults.NotFound() : TypedResults.Ok(urun);
    }
}

// UrunEndpointsTests.cs  (xunit + Microsoft.EntityFrameworkCore.InMemory)
using Microsoft.AspNetCore.Http.HttpResults;
using Microsoft.EntityFrameworkCore;
using Xunit;

public class UrunEndpointsTests
{
    private static AppDb YeniDb() =>
        new(new DbContextOptionsBuilder<AppDb>()
            .UseInMemoryDatabase(Guid.NewGuid().ToString()).Options);

    [Fact]
    public async Task Getir_VarOlanId_200_ve_Urun()
    {
        using var db = YeniDb();
        db.Urunler.Add(new Urun(1, "Kalem", 9.5m));
        await db.SaveChangesAsync();

        var sonuc = await UrunEndpoints.Getir(1, db);

        var ok = Assert.IsType<Ok<Urun>>(sonuc.Result);
        Assert.Equal("Kalem", ok.Value!.Ad);
    }

    [Fact]
    public async Task Getir_OlmayanId_404()
    {
        using var db = YeniDb();

        var sonuc = await UrunEndpoints.Getir(42, db);

        Assert.IsType<NotFound>(sonuc.Result);
    }
}
