public class Urun
{
    public int Id { get; set; }

**UrunEndpointTests.cs**
```csharp
using System.Net;
using System.Net.Http.Json;
using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;
using Xunit;

public class UrunEndpointTests : IDisposable
{
    private readonly WebApplicationFactory<Program> _factory;
    private readonly HttpClient _client;

    public UrunEndpointTests()
    {
        var dbAdi = Guid.NewGuid().ToString();

        _factory = new WebApplicationFactory<Program>().WithWebHostBuilder(b =>
            b.ConfigureTestServices(s =>
            {
                s.RemoveAll<DbContextOptions<UrunDb>>();
                s.AddDbContext<UrunDb>(o => o.UseInMemoryDatabase(dbAdi));
            }));

        using var scope = _factory.Services.CreateScope();
        var db = scope.ServiceProvider.GetRequiredService<UrunDb>();
        db.Urunler.Add(new Urun { Id = 1, Ad = "Kalem", Fiyat = 12.5m });
        db.SaveChanges();

        _client = _factory.CreateClient();
    }

    [Fact]
    public async Task GetUrun_VarOlanId_200VeUrunDoner()
    {
        var response = await _client.GetAsync("/urunler/1");

        Assert.Equal(HttpStatusCode.OK, response.StatusCode);
        var urun = await response.Content.ReadFromJsonAsync<Urun>();
        Assert.NotNull(urun);
        Assert.Equal("Kalem", urun.Ad);
    }

    [Fact]
    public async Task GetUrun_OlmayanId_404Doner()
    {
        var response = await _client.GetAsync("/urunler/999");

        Assert.Equal(HttpStatusCode.NotFound, response.StatusCode);
    }

    public void Dispose()
    {
        _client.Dispose();
        _factory.Dispose();
    }
}
```
