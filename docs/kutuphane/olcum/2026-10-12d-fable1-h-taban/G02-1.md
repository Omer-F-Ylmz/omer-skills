builder.Services.AddDbContext<AppDbContext>(o => o.UseInMemoryDatabase("urunler"));

var app = builder.Build();

app.MapGet("/urunler/{id:int}", async (int id, AppDbContext db) =>

app.Run();

// UrunEndpointTests.cs
using System.Net;
using System.Net.Http.Json;
using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.Extensions.DependencyInjection;
using Xunit;

public class UrunEndpointTests : IClassFixture<WebApplicationFactory<Program>>
{
    private readonly HttpClient _client;

    public UrunEndpointTests(WebApplicationFactory<Program> factory)
    {
        using var scope = factory.Services.CreateScope();
        var db = scope.ServiceProvider.GetRequiredService<AppDbContext>();
        db.Urunler.Add(new Urun(1, "Kalem", 12.5m));
        db.SaveChanges();

        _client = factory.CreateClient();
    }

    [Fact]
    public async Task GetUrun_VarOlanId_200VeUrunDoner()
    {
        var urun = await _client.GetFromJsonAsync<Urun>("/urunler/1");

        Assert.NotNull(urun);
        Assert.Equal("Kalem", urun.Ad);
    }

    [Fact]
    public async Task GetUrun_OlmayanId_404Doner()
    {
        var response = await _client.GetAsync("/urunler/999");

        Assert.Equal(HttpStatusCode.NotFound, response.StatusCode);
    }
}
