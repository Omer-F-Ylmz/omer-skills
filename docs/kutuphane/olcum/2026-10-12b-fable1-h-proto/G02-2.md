**Program.cs** (web projesi)

```csharp
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddDbContext<AppDbContext>(o =>
    o.UseSqlServer(builder.Configuration.GetConnectionString("Default")));

var app = builder.Build();

app.MapGet("/urunler/{id:int}", async (int id, AppDbContext db) =>
    await db.Urunler.AsNoTracking().FirstOrDefaultAsync(u => u.Id == id) is { } urun
        ? Results.Ok(urun)
        : Results.NotFound());

app.Run();

public partial class Program { }

public class Urun
{
    public int Id { get; set; }
    public string Ad { get; set; } = "";
    public decimal Fiyat { get; set; }
}

public class AppDbContext(DbContextOptions<AppDbContext> options) : DbContext(options)
{
    public DbSet<Urun> Urunler => Set<Urun>();
}
```

**UrunEndpointTests.cs** (test projesi)

```csharp
using System.Net;
using System.Net.Http.Json;
using Microsoft.AspNetCore.Hosting;
using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;
using Xunit;

public class TestFabrika : WebApplicationFactory<Program>
{
    private readonly string _dbAdi = Guid.NewGuid().ToString();

    protected override void ConfigureWebHost(IWebHostBuilder builder)
    {
        builder.ConfigureServices(services =>
        {
            services.RemoveAll<DbContextOptions<AppDbContext>>();
            services.AddDbContext<AppDbContext>(o => o.UseInMemoryDatabase(_dbAdi));
        });
    }
}

public class UrunEndpointTests : IClassFixture<TestFabrika>
{
    private readonly HttpClient _istemci;

    public UrunEndpointTests(TestFabrika fabrika)
    {
        using var scope = fabrika.Services.CreateScope();
        var db = scope.ServiceProvider.GetRequiredService<AppDbContext>();
        db.Database.EnsureCreated();
        if (!db.Urunler.Any())
        {
            db.Urunler.Add(new Urun { Id = 1, Ad = "Kalem", Fiyat = 12.5m });
            db.SaveChanges();
        }

        _istemci = fabrika.CreateClient();
    }

    [Fact]
    public async Task GetUrun_VarOlanId_200VeUrunDoner()
    {
        var yanit = await _istemci.GetAsync("/urunler/1");

        Assert.Equal(HttpStatusCode.OK, yanit.StatusCode);
        var urun = await yanit.Content.ReadFromJsonAsync<Urun>();
        Assert.NotNull(urun);
        Assert.Equal("Kalem", urun.Ad);
    }

    [Fact]
    public async Task GetUrun_YokId_404Doner()
    {
        var yanit = await _istemci.GetAsync("/urunler/999");

        Assert.Equal(HttpStatusCode.NotFound, yanit.StatusCode);
    }
}
```
