// Program.cs
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddDbContext<AppDb>(o => o.UseInMemoryDatabase("urunler"));
var app = builder.Build();

app.MapGet("/urunler/{id:int}", async (int id, AppDb db) =>
    await db.Urunler.FindAsync(id) is Urun u ? Results.Ok(u) : Results.NotFound());

app.Run();

public record Urun(int Id, string Ad, decimal Fiyat);

public class AppDb(DbContextOptions<AppDb> o) : DbContext(o)
{
    public DbSet<Urun> Urunler => Set<Urun>();
}

```csharp
// UrunTests.cs
// Paketler: Microsoft.AspNetCore.Mvc.Testing, Microsoft.EntityFrameworkCore.InMemory, xunit
using System.Net;
using System.Net.Http.Json;
using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;
using Xunit;

public class UrunTests : IClassFixture<WebApplicationFactory<Program>>
{
    private readonly WebApplicationFactory<Program> _factory;

    public UrunTests(WebApplicationFactory<Program> factory) =>
        _factory = factory.WithWebHostBuilder(b => b.ConfigureServices(s =>
        {
            s.RemoveAll<DbContextOptions<AppDb>>();
            s.AddDbContext<AppDb>(o => o.UseInMemoryDatabase(Guid.NewGuid().ToString()));
        }));

    [Fact]
    public async Task Get_Mevcut_Urun_200_Doner()
    {
        using (var scope = _factory.Services.CreateScope())
        {
            var db = scope.ServiceProvider.GetRequiredService<AppDb>();
            db.Urunler.Add(new Urun(1, "Kalem", 12.5m));
            await db.SaveChangesAsync();
        }

        var res = await _factory.CreateClient().GetAsync("/urunler/1");

        Assert.Equal(HttpStatusCode.OK, res.StatusCode);
        Assert.Equal(new Urun(1, "Kalem", 12.5m), await res.Content.ReadFromJsonAsync<Urun>());
    }

    [Fact]
    public async Task Get_Olmayan_Urun_404_Doner()
    {
        var res = await _factory.CreateClient().GetAsync("/urunler/999");

        Assert.Equal(HttpStatusCode.NotFound, res.StatusCode);
    }
}
```

`RemoveAll` için `using Microsoft.Extensions.DependencyInjection.Extensions;` gerekir. Testleri çalıştırmadım. Program.cs'te de InMemory kullandım; gerçek DB'ye geçince test override'ı aynen kalır.
