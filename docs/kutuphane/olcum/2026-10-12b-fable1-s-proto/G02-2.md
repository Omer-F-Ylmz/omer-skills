```csharp
// Program.cs
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddDbContext<AppDb>(o => o.UseInMemoryDatabase("urunler"));
var app = builder.Build();

app.MapGet("/urunler/{id:int}", async (int id, AppDb db) =>
    await db.Urunler.FindAsync(id) is Urun u ? Results.Ok(u) : Results.NotFound());

app.Run();

public record Urun(int Id, string Ad, decimal Fiyat);

public class AppDb(DbContextOptions<AppDb> options) : DbContext(options)
{
    public DbSet<Urun> Urunler => Set<Urun>();
}

public partial class Program { } // WebApplicationFactory için
```

```csharp
// UrunTests.cs
// Test projesi paketleri: xunit, Microsoft.AspNetCore.Mvc.Testing; web projesine ProjectReference
using System.Net;
using System.Net.Http.Json;
using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.Extensions.DependencyInjection;
using Xunit;

public class UrunTests(WebApplicationFactory<Program> factory)
    : IClassFixture<WebApplicationFactory<Program>>
{
    [Fact]
    public async Task Var_olan_urun_200_doner()
    {
        using (var scope = factory.Services.CreateScope())
        {
            var db = scope.ServiceProvider.GetRequiredService<AppDb>();
            db.Urunler.Add(new Urun(1, "Kalem", 12.5m));
            await db.SaveChangesAsync();
        }

        var res = await factory.CreateClient().GetAsync("/urunler/1");

        Assert.Equal(HttpStatusCode.OK, res.StatusCode);
        var urun = await res.Content.ReadFromJsonAsync<Urun>();
        Assert.Equal("Kalem", urun!.Ad);
    }

    [Fact]
    public async Task Olmayan_urun_404_doner()
    {
        var res = await factory.CreateClient().GetAsync("/urunler/999");

        Assert.Equal(HttpStatusCode.NotFound, res.StatusCode);
    }
}
```

Bunları çalıştırmadım. Test, uygulamanın kendi InMemory DB'sini kullanıyor (`DbContext` değiştirmedim), o yüzden id'ler testler arasında çakışmasın diye 1 ve 999 sabit.
