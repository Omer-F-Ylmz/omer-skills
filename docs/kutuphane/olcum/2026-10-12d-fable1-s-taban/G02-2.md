// Program.cs
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddDbContext<UrunDb>(o => o.UseInMemoryDatabase("urunler"));
var app = builder.Build();

app.MapGet("/urunler/{id:int}", async (int id, UrunDb db) =>
    await db.Urunler.FindAsync(id) is Urun u ? Results.Ok(u) : Results.NotFound());

app.Run();

public record Urun(int Id, string Ad, decimal Fiyat);

public class UrunDb(DbContextOptions<UrunDb> options) : DbContext(options)
{
    public DbSet<Urun> Urunler => Set<Urun>();
}

// UrunEndpointTests.cs
// dotnet add package Microsoft.AspNetCore.Mvc.Testing
// dotnet add package Microsoft.EntityFrameworkCore.InMemory
using System.Net;
using System.Net.Http.Json;
using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.Extensions.DependencyInjection;
using Xunit;

public class UrunEndpointTests(WebApplicationFactory<Program> factory)
    : IClassFixture<WebApplicationFactory<Program>>
{
    [Fact]
    public async Task Var_Olan_Urun_200_Doner()
    {
        using (var scope = factory.Services.CreateScope())
        {
            var db = scope.ServiceProvider.GetRequiredService<UrunDb>();
            db.Urunler.Add(new Urun(1, "Kalem", 12.5m));
            await db.SaveChangesAsync();
        }

        var res = await factory.CreateClient().GetAsync("/urunler/1");

        Assert.Equal(HttpStatusCode.OK, res.StatusCode);
        var urun = await res.Content.ReadFromJsonAsync<Urun>();
        Assert.Equal("Kalem", urun!.Ad);
    }

    [Fact]
    public async Task Olmayan_Urun_404_Doner()
    {
        var res = await factory.CreateClient().GetAsync("/urunler/999");

        Assert.Equal(HttpStatusCode.NotFound, res.StatusCode);
    }
}
