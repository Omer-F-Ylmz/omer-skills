Tek dosya. Kaydedip tarayıcıda açman yeterli, sunucu gerekmiyor.

```html
<!doctype html>
<html>
<head>
<meta charset="utf-8">
<style>body{margin:0;overflow:hidden}</style>
<script type="importmap">
{ "imports": { "three": "https://cdn.jsdelivr.net/npm/three@0.170.0/build/three.module.js" } }
</script>
</head>
<body>
<script type="module">
import * as THREE from 'three';

const scene = new THREE.Scene();
scene.background = new THREE.Color(0x202025);
const camera = new THREE.PerspectiveCamera(60, innerWidth / innerHeight, 0.1, 1000);
const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(innerWidth, innerHeight);
document.body.appendChild(renderer.domElement);

scene.add(new THREE.HemisphereLight(0xffffff, 0x444444, 1.5));
scene.add(new THREE.GridHelper(100, 100));

const cube = new THREE.Mesh(new THREE.BoxGeometry(), new THREE.MeshStandardMaterial({ color: 0x44aaff }));
cube.position.y = 0.5;
scene.add(cube);

// e.code fiziksel tuşu verir, klavye düzeni ne olursa olsun WASD çalışır
const keys = {};
addEventListener('keydown', e => keys[e.code] = true);
addEventListener('keyup', e => keys[e.code] = false);

const speed = 5;
const offset = new THREE.Vector3(0, 4, 8); // kameranın küpe göre konumu
const dir = new THREE.Vector3();
const target = new THREE.Vector3();
const clock = new THREE.Clock();
camera.position.copy(cube.position).add(offset);

renderer.setAnimationLoop(() => {
  const dt = clock.getDelta();

  dir.set((keys.KeyD ? 1 : 0) - (keys.KeyA ? 1 : 0), 0,
          (keys.KeyS ? 1 : 0) - (keys.KeyW ? 1 : 0));
  if (dir.lengthSq()) cube.position.addScaledVector(dir.normalize(), speed * dt);

  // Kare hızından bağımsız yumuşak takip
  target.copy(cube.position).add(offset);
  camera.position.lerp(target, 1 - Math.exp(-5 * dt));
  camera.lookAt(cube.position);

  renderer.render(scene, camera);
});

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
</script>
</body>
</html>
```

WASD dünya eksenlerine göre hareket ettiriyor, küp dönmüyor. Kameranın arkasından gelen hareket, yani küpü döndürüp yerel eksende ilerletme isterseniz söyleyin; birkaç satırlık bir ek.
