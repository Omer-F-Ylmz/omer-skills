// Api/Program.cs
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddDbContext<AppDb>(o => o.UseInMemoryDatabase("urunler"));
var app = builder.Build();

app.MapGet("/urunler/{id:int}", async (int id, AppDb db) =>
    await db.Urunler.FindAsync(id) is Urun urun ? Results.Ok(urun) : Results.NotFound());

app.Run();

public record Urun(int Id, string Ad);

public class AppDb(DbContextOptions<AppDb> options) : DbContext(options)
{
    public DbSet<Urun> Urunler => Set<Urun>();
}

public partial class Program;

// Api.Tests/UrunlerTests.cs
// paketler: xunit, xunit.runner.visualstudio, Microsoft.NET.Test.Sdk, Microsoft.AspNetCore.Mvc.Testing
using System.Net;
using System.Net.Http.Json;
using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.Extensions.DependencyInjection;

public class UrunlerTests(WebApplicationFactory<Program> factory) : IClassFixture<WebApplicationFactory<Program>>
{
    [Fact]
    public async Task Var_olan_urun_200_ve_urunu_doner()
    {
        using (var scope = factory.Services.CreateScope())
        {
            var db = scope.ServiceProvider.GetRequiredService<AppDb>();
            db.Urunler.Add(new Urun(1, "Kalem"));
            await db.SaveChangesAsync();
        }

        var yanit = await factory.CreateClient().GetAsync("/urunler/1");

        Assert.Equal(HttpStatusCode.OK, yanit.StatusCode);
        Assert.Equal(new Urun(1, "Kalem"), await yanit.Content.ReadFromJsonAsync<Urun>());
    }

    [Fact]
    public async Task Olmayan_urun_404_doner()
    {
        var yanit = await factory.CreateClient().GetAsync("/urunler/999");

        Assert.Equal(HttpStatusCode.NotFound, yanit.StatusCode);
    }
}
