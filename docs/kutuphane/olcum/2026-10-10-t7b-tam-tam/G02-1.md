**Api/Program.cs**
```csharp
using Microsoft.AspNetCore.Http.HttpResults;
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddDbContext<AppDbContext>(o => o.UseInMemoryDatabase("urunler"));

var app = builder.Build();
app.MapGet("/urunler/{id:int}", UrunEndpoints.GetById);
app.Run();

public class Urun
{
    public int Id { get; set; }
    public string Ad { get; set; } = "";
    public decimal Fiyat { get; set; }
}

public class AppDbContext(DbContextOptions<AppDbContext> options) : DbContext(options)
{
    public DbSet<Urun> Urunler => Set<Urun>();
}

public static class UrunEndpoints
{
    public static async Task<Results<Ok<Urun>, NotFound>> GetById(int id, AppDbContext db) =>
        await db.Urunler.FindAsync(id) is { } urun
            ? TypedResults.Ok(urun)
            : TypedResults.NotFound();
}
```

**Api.Tests/UrunEndpointsTests.cs**
```csharp
using Microsoft.AspNetCore.Http.HttpResults;
using Microsoft.EntityFrameworkCore;

public class UrunEndpointsTests
{
    static AppDbContext YeniDb() => new(new DbContextOptionsBuilder<AppDbContext>()
        .UseInMemoryDatabase(Guid.NewGuid().ToString())
        .Options);

    [Fact]
    public async Task Var_olan_urun_200_ve_urunu_doner()
    {
        using var db = YeniDb();
        db.Urunler.Add(new Urun { Id = 1, Ad = "Kalem", Fiyat = 10m });
        await db.SaveChangesAsync();

        var sonuc = await UrunEndpoints.GetById(1, db);

        var ok = Assert.IsType<Ok<Urun>>(sonuc.Result);
        Assert.Equal("Kalem", ok.Value!.Ad);
    }

    [Fact]
    public async Task Olmayan_urun_404_doner()
    {
        using var db = YeniDb();

        var sonuc = await UrunEndpoints.GetById(99, db);

        Assert.IsType<NotFound>(sonuc.Result);
    }
}
```
