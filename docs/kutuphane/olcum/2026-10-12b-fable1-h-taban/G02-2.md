```csharp
// Models.cs
using Microsoft.EntityFrameworkCore;

public record Urun(int Id, string Ad, decimal Fiyat);

public class AppDbContext(DbContextOptions<AppDbContext> options) : DbContext(options)
{
    public DbSet<Urun> Urunler => Set<Urun>();
}
```

```csharp
// Endpoints/UrunEndpoints.cs
using Microsoft.AspNetCore.Http.HttpResults;

public static class UrunEndpoints
{
    public static async Task<Results<Ok<Urun>, NotFound>> GetById(int id, AppDbContext db) =>
        await db.Urunler.FindAsync(id) is { } urun
            ? TypedResults.Ok(urun)
            : TypedResults.NotFound();
}
```

```csharp
// Program.cs
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);

// ponytail: InMemory, gerçek sağlayıcıya geçince burayı değiştir
builder.Services.AddDbContext<AppDbContext>(o => o.UseInMemoryDatabase("urunler"));

var app = builder.Build();

app.MapGet("/urunler/{id:int}", UrunEndpoints.GetById);

app.Run();
```

```csharp
// Tests/UrunEndpointsTests.cs
using Microsoft.AspNetCore.Http.HttpResults;
using Microsoft.EntityFrameworkCore;
using Xunit;

public class UrunEndpointsTests
{
    private static AppDbContext CreateDb(string name)
    {
        var options = new DbContextOptionsBuilder<AppDbContext>()
            .UseInMemoryDatabase(name)
            .Options;
        return new AppDbContext(options);
    }

    [Fact]
    public async Task GetById_VarOlanUrun_OkDoner()
    {
        using var db = CreateDb(nameof(GetById_VarOlanUrun_OkDoner));
        db.Urunler.Add(new Urun(1, "Kalem", 12.5m));
        await db.SaveChangesAsync();

        var result = await UrunEndpoints.GetById(1, db);

        var ok = Assert.IsType<Ok<Urun>>(result.Result);
        Assert.Equal("Kalem", ok.Value!.Ad);
    }

    [Fact]
    public async Task GetById_YokUrun_NotFoundDoner()
    {
        using var db = CreateDb(nameof(GetById_YokUrun_NotFoundDoner));

        var result = await UrunEndpoints.GetById(99, db);

        Assert.IsType<NotFound>(result.Result);
    }
}
```
