WASD ile hareket eden küp ve onu yumuşak şekilde takip eden kamera. Tek HTML dosyası, kurulum gerekmez:

```html
<!doctype html>
<html>
<head>
<meta charset="utf-8">
<title>WASD küp</title>
<style>body{margin:0;overflow:hidden}</style>
<script type="importmap">
{ "imports": { "three": "https://unpkg.com/three@0.170.0/build/three.module.js" } }
</script>
</head>
<body>
<script type="module">
import * as THREE from 'three';

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(innerWidth, innerHeight);
document.body.appendChild(renderer.domElement);

const scene = new THREE.Scene();
scene.background = new THREE.Color(0x202025);
scene.add(new THREE.HemisphereLight(0xffffff, 0x444444, 2));
scene.add(new THREE.GridHelper(100, 100));

const cube = new THREE.Mesh(
  new THREE.BoxGeometry(1, 1, 1),
  new THREE.MeshStandardMaterial({ color: 0x44aaff })
);
cube.position.y = 0.5;
scene.add(cube);

const camera = new THREE.PerspectiveCamera(60, innerWidth / innerHeight, 0.1, 1000);
const offset = new THREE.Vector3(0, 4, 8);           // küpe göre kamera konumu
camera.position.copy(cube.position).add(offset);

const keys = {};
addEventListener('keydown', e => keys[e.code] = true);
addEventListener('keyup',   e => keys[e.code] = false);

const clock = new THREE.Clock();
const dir = new THREE.Vector3();
const camTarget = new THREE.Vector3();
const speed = 5;

renderer.setAnimationLoop(() => {
  const dt = clock.getDelta();

  dir.set(
    (keys.KeyD ? 1 : 0) - (keys.KeyA ? 1 : 0),
    0,
    (keys.KeyS ? 1 : 0) - (keys.KeyW ? 1 : 0)
  );
  if (dir.lengthSq()) cube.position.addScaledVector(dir.normalize(), speed * dt);

  camTarget.copy(cube.position).add(offset);
  camera.position.lerp(camTarget, 1 - Math.exp(-5 * dt)); // fps'ten bağımsız yumuşatma
  camera.lookAt(cube.position);

  renderer.render(scene, camera);
});

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
</script>
</body>
</html>
```

- Tuşlar `e.code` ile okunuyor, yani klavyedeki fiziksel konuma bakılıyor. Türkçe Q ve F klavyede de aynı yerdeki tuşlar çalışır.
- Kamera küpün dönüşünü takip etmez, hep aynı sabit açıdan bakar. Küpün kendi yönüne göre hareket istersen `offset.clone().applyQuaternion(cube.quaternion)` kullan ve A/D tuşlarını dönmeye bağla.
