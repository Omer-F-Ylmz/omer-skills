// Api/Program.cs  (paket: Microsoft.EntityFrameworkCore.InMemory)
using Microsoft.AspNetCore.Http.HttpResults;
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddDbContext<AppDb>(o => o.UseInMemoryDatabase("urunler"));
var app = builder.Build();

app.MapGet("/urunler/{id:int}", UrunEndpoints.GetById);

app.Run();

public class Urun
{
    public int Id { get; set; }
    public string Ad { get; set; } = "";
    public decimal Fiyat { get; set; }
}

public class AppDb(DbContextOptions<AppDb> options) : DbContext(options)
{
    public DbSet<Urun> Urunler => Set<Urun>();
}

public static class UrunEndpoints
{
    public static async Task<Results<Ok<Urun>, NotFound>> GetById(int id, AppDb db) =>
        await db.Urunler.FindAsync(id) is { } urun ? TypedResults.Ok(urun) : TypedResults.NotFound();
}

// Api.Tests/UrunEndpointsTests.cs  (xunit + ProjectReference ../Api)
using Microsoft.AspNetCore.Http.HttpResults;
using Microsoft.EntityFrameworkCore;
using Xunit;

public class UrunEndpointsTests
{
    static AppDb YeniDb() => new(new DbContextOptionsBuilder<AppDb>()
        .UseInMemoryDatabase(Guid.NewGuid().ToString()).Options);

    [Fact]
    public async Task Urun_varsa_200_ve_urunu_doner()
    {
        await using var db = YeniDb();
        db.Urunler.Add(new Urun { Id = 1, Ad = "Kalem", Fiyat = 10m });
        await db.SaveChangesAsync();

        var sonuc = await UrunEndpoints.GetById(1, db);

        var ok = Assert.IsType<Ok<Urun>>(sonuc.Result);
        Assert.Equal("Kalem", ok.Value!.Ad);
    }

    [Fact]
    public async Task Urun_yoksa_404_doner()
    {
        await using var db = YeniDb();

        var sonuc = await UrunEndpoints.GetById(99, db);

        Assert.IsType<NotFound>(sonuc.Result);
    }
}
