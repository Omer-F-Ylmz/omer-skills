**Program.cs**
```csharp
using Microsoft.AspNetCore.Http.HttpResults;
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddDbContext<AppDb>(o => o.UseInMemoryDatabase("urunler"));
var app = builder.Build();

app.MapGet("/urunler/{id:int}", UrunEndpoints.Getir);

app.Run();

public class Urun
{
    public int Id { get; set; }
    public string Ad { get; set; } = "";
    public decimal Fiyat { get; set; }
}

public class AppDb(DbContextOptions<AppDb> options) : DbContext(options)
{
    public DbSet<Urun> Urunler => Set<Urun>();
}

public static class UrunEndpoints
{
    public static async Task<Results<Ok<Urun>, NotFound>> Getir(int id, AppDb db) =>
        await db.Urunler.FindAsync(id) is { } urun
            ? TypedResults.Ok(urun)
            : TypedResults.NotFound();
}
```

**Tests/UrunEndpointsTests.cs**
```csharp
// Paketler: xunit, xunit.runner.visualstudio, Microsoft.NET.Test.Sdk,
// Microsoft.EntityFrameworkCore.InMemory + API projesine ProjectReference
using Microsoft.AspNetCore.Http.HttpResults;
using Microsoft.EntityFrameworkCore;
using Xunit;

public class UrunEndpointsTests
{
    private static AppDb YeniDb()
    {
        var options = new DbContextOptionsBuilder<AppDb>()
            .UseInMemoryDatabase(Guid.NewGuid().ToString())
            .Options;
        var db = new AppDb(options);
        db.Urunler.Add(new Urun { Id = 1, Ad = "Kalem", Fiyat = 10m });
        db.SaveChanges();
        return db;
    }

    [Fact]
    public async Task Var_olan_urun_200_doner()
    {
        using var db = YeniDb();

        var sonuc = await UrunEndpoints.Getir(1, db);

        var ok = Assert.IsType<Ok<Urun>>(sonuc.Result);
        Assert.Equal("Kalem", ok.Value!.Ad);
    }

    [Fact]
    public async Task Olmayan_urun_404_doner()
    {
        using var db = YeniDb();

        var sonuc = await UrunEndpoints.Getir(99, db);

        Assert.IsType<NotFound>(sonuc.Result);
    }
}
```
