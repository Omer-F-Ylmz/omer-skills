Tek dosya. Kaydedip tarayıcıda açman yeterli; importmap `file://` üzerinden de çalışır.

```html
<!doctype html>
<html>
<head>
<meta charset="utf-8">
<style>body{margin:0;overflow:hidden}</style>
<script type="importmap">
{ "imports": { "three": "https://cdn.jsdelivr.net/npm/three@0.170.0/build/three.module.js" } }
</script>
</head>
<body>
<script type="module">
import * as THREE from 'three';

const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(60, innerWidth / innerHeight, 0.1, 100);
const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(innerWidth, innerHeight);
document.body.appendChild(renderer.domElement);

scene.add(new THREE.HemisphereLight(0xffffff, 0x444444, 2));
scene.add(new THREE.GridHelper(50, 50));

const cube = new THREE.Mesh(
  new THREE.BoxGeometry(),
  new THREE.MeshStandardMaterial({ color: 0x44aa88 })
);
cube.position.y = 0.5;
scene.add(cube);

const keys = {};
addEventListener('keydown', e => keys[e.code] = true);
addEventListener('keyup',   e => keys[e.code] = false);

const offset = new THREE.Vector3(0, 3, 6);   // kameranın küpe göre konumu
const speed = 5;                              // birim/sn
const dir = new THREE.Vector3();
camera.position.copy(cube.position).add(offset);

let last = 0;
renderer.setAnimationLoop(t => {
  const dt = Math.min((t - last) / 1000, 0.1); last = t;  // sekme değişiminde sıçramayı önler

  dir.set((keys.KeyD ? 1 : 0) - (keys.KeyA ? 1 : 0), 0,
          (keys.KeyS ? 1 : 0) - (keys.KeyW ? 1 : 0)).normalize();  // çaprazda hız artmasın
  cube.position.addScaledVector(dir, speed * dt);

  const target = cube.position.clone().add(offset);
  camera.position.lerp(target, 1 - Math.exp(-5 * dt));     // FPS'ten bağımsız yumuşak takip
  camera.lookAt(cube.position);

  renderer.render(scene, camera);
});

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
</script>
</body>
</html>
```

- Hareket dünya eksenlerinde: W ileri (-Z), A sola. Küpü A/D ile döndürüp kamerayı arkasında tutmak istersen kodu ona göre değiştiririm.
- `e.code` tuşun klavyedeki fiziksel yerine bakar, bu yüzden Türkçe Q ve F klavyede de çalışır.
- Kodu tarayıcıda çalıştırmadım, test edilmedi.
