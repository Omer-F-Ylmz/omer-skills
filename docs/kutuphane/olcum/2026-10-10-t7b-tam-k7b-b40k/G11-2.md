Tek dosya, kurulum gerekmez. `index.html` olarak kaydedip tarayıcıda açman yeterli:

```html
<!doctype html>
<html>
<head>
<meta charset="utf-8">
<style>body{margin:0;overflow:hidden}</style>
<script type="importmap">
{ "imports": { "three": "https://unpkg.com/three@0.170.0/build/three.module.js" } }
</script>
</head>
<body>
<script type="module">
import * as THREE from 'three';

const scene = new THREE.Scene();
scene.background = new THREE.Color(0x87ceeb);
const camera = new THREE.PerspectiveCamera(60, innerWidth / innerHeight, 0.1, 500);
const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(innerWidth, innerHeight);
document.body.appendChild(renderer.domElement);

scene.add(new THREE.HemisphereLight(0xffffff, 0x444444, 2));
scene.add(new THREE.GridHelper(200, 100));

const cube = new THREE.Mesh(
  new THREE.BoxGeometry(1, 1, 1),
  new THREE.MeshStandardMaterial({ color: 0xff5533 })
);
cube.position.y = 0.5;
scene.add(cube);

const keys = {};
addEventListener('keydown', e => keys[e.code] = true);
addEventListener('keyup',   e => keys[e.code] = false);

const offset = new THREE.Vector3(0, 3, 6);   // küpün arkası + üstü (yerel uzay)
const camGoal = new THREE.Vector3();
const clock = new THREE.Clock();
camera.position.copy(offset);

renderer.setAnimationLoop(() => {
  const dt = clock.getDelta();

  // A/D döndürür, W/S küpün baktığı yönde ileri/geri götürür
  if (keys.KeyA) cube.rotation.y += 2 * dt;
  if (keys.KeyD) cube.rotation.y -= 2 * dt;
  cube.translateZ(((keys.KeyS ? 1 : 0) - (keys.KeyW ? 1 : 0)) * 5 * dt);

  // Kamera: hedef noktayı küple birlikte döndür, oraya yumuşak geçiş yap, küpe bak
  camGoal.copy(offset).applyQuaternion(cube.quaternion).add(cube.position);
  camera.position.lerp(camGoal, 1 - Math.exp(-5 * dt));   // FPS'ten bağımsız yumuşatma
  camera.lookAt(cube.position);

  renderer.render(scene, camera);
});

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
</script>
</body>
</html>
```

- Tuşlar `e.code` ile okunuyor. Bu kod fiziksel tuşa bakar, yani Türkçe Q klavyede de çalışır.
- Kamera her zaman küpün arkasında kalır. Küpün dönüşünü takip etmesin, dünyaya göre sabit dursun istersen `.applyQuaternion(cube.quaternion)` kısmını sil.
- Çarpışma ve fizik eklemedim. Gerekirse sonra eklenebilir.
