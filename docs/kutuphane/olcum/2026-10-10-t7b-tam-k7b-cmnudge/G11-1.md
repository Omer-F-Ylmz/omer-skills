Kod tek dosya. Bir `.html` dosyasına kaydedip tarayıcıda açın. Dosya çift tıkla da açılır, sunucu gerekmez.

```html
<!doctype html>
<html>
<head>
<meta charset="utf-8">
<style>body{margin:0;overflow:hidden}</style>
<script type="importmap">
{ "imports": { "three": "https://cdn.jsdelivr.net/npm/three@0.170.0/build/three.module.js" } }
</script>
</head>
<body>
<script type="module">
import * as THREE from 'three';

const scene = new THREE.Scene();
scene.background = new THREE.Color(0x202025);
const camera = new THREE.PerspectiveCamera(60, innerWidth / innerHeight, 0.1, 200);
const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(innerWidth, innerHeight);
document.body.appendChild(renderer.domElement);

scene.add(new THREE.HemisphereLight(0xffffff, 0x444444, 2));
scene.add(new THREE.GridHelper(100, 100));

const cube = new THREE.Mesh(new THREE.BoxGeometry(), new THREE.MeshStandardMaterial({ color: 0x44aaff }));
cube.position.y = 0.5;
scene.add(cube);

// e.code klavye düzeninden bağımsızdır (Türkçe Q/F'de de W fiziksel tuşu)
const keys = new Set();
addEventListener('keydown', e => keys.add(e.code));
addEventListener('keyup', e => keys.delete(e.code));
addEventListener('blur', () => keys.clear()); // pencere odağı kaybolunca tuş takılı kalmasın

const offset = new THREE.Vector3(0, 4, 8); // kamera küpün arkasında ve üstünde
const dir = new THREE.Vector3();
const goal = new THREE.Vector3();
const clock = new THREE.Clock();
camera.position.copy(cube.position).add(offset);

renderer.setAnimationLoop(() => {
  const dt = clock.getDelta();

  dir.set(keys.has('KeyD') - keys.has('KeyA'), 0, keys.has('KeyS') - keys.has('KeyW'));
  if (dir.lengthSq()) cube.position.addScaledVector(dir.normalize(), 5 * dt); // çaprazda hızlanmasın

  goal.copy(cube.position).add(offset);
  camera.position.lerp(goal, 1 - Math.exp(-5 * dt)); // FPS'ten bağımsız yumuşak takip
  camera.lookAt(cube.position);

  renderer.render(scene, camera);
});

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
</script>
</body>
</html>
```

Hareket dünya eksenlerine göre. W ileri (-Z), A sola gider. Takip hızını `-5 * dt` içindeki 5 belirler: değeri büyütürseniz kamera daha sıkı takip eder.

Küpün kendi etrafında dönmesini (A/D ile dönme) ve kameranın hep arkasında kalmasını eklemedim. İsterseniz söyleyin, yaklaşık 5 satır ekler. Kodu tarayıcıda çalıştırmadım.
