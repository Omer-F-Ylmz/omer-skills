Tek dosya, kurulum gerekmez. `index.html` olarak kaydedip tarayıcıda aç:

```html
<!doctype html>
<html lang="tr">
<head>
  <meta charset="utf-8">
  <title>WASD küp + takip kamerası</title>
  <style>body { margin: 0; overflow: hidden; }</style>
  <script type="importmap">
    { "imports": { "three": "https://unpkg.com/three@0.170.0/build/three.module.js" } }
  </script>
</head>
<body>
<script type="module">
import * as THREE from 'three';

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(innerWidth, innerHeight);
document.body.appendChild(renderer.domElement);

const scene = new THREE.Scene();
scene.background = new THREE.Color(0x202025);
scene.add(new THREE.HemisphereLight(0xffffff, 0x444444, 2));
scene.add(new THREE.GridHelper(100, 100));

const camera = new THREE.PerspectiveCamera(60, innerWidth / innerHeight, 0.1, 500);

const cube = new THREE.Mesh(
  new THREE.BoxGeometry(1, 1, 1),
  new THREE.MeshStandardMaterial({ color: 0x44aa88 })
);
cube.position.y = 0.5;
scene.add(cube);

// e.code fiziksel tuşu verir, Türkçe klavye düzeninde de çalışır
const keys = {};
addEventListener('keydown', e => keys[e.code] = true);
addEventListener('keyup',   e => keys[e.code] = false);

const SPEED = 5;                            // birim/sn
const offset = new THREE.Vector3(0, 4, 8);  // kameranın küpe göre konumu
const dir = new THREE.Vector3();
const camTarget = new THREE.Vector3();
const clock = new THREE.Clock();

renderer.setAnimationLoop(() => {
  const dt = clock.getDelta();

  dir.set(
    (keys.KeyD ? 1 : 0) - (keys.KeyA ? 1 : 0),
    0,
    (keys.KeyS ? 1 : 0) - (keys.KeyW ? 1 : 0)
  );
  if (dir.lengthSq() > 0) cube.position.addScaledVector(dir.normalize(), SPEED * dt);

  // Kamera hedefe yumuşakça yaklaşır; kare hızından bağımsızdır
  camTarget.copy(cube.position).add(offset);
  camera.position.lerp(camTarget, 1 - Math.exp(-5 * dt));
  camera.lookAt(cube.position);

  renderer.render(scene, camera);
});

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});
</script>
</body>
</html>
```

- Küp dünya eksenlerinde hareket eder. Çapraz giderken de hızı aynı kalır çünkü yön vektörü normalize ediliyor.
- Kamera her zaman küpün arkasında ve üstünde, sabit bir mesafede durur. `exp` ile yapılan lerp, takibin kare hızından bağımsız olmasını sağlıyor.
- Tarayıcıda çalıştırıp test etmedim.

İstersen küpü tank gibi de sürdürebilirim: A/D küpü döndürür, W/S kendi yönünde ileri-geri götürür. Bunun için `offset`'i `cube.quaternion` ile döndürmek yeterli.
