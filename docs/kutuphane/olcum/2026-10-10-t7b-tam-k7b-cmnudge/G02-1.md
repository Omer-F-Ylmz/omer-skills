**Api/Program.cs**
```csharp
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddDbContext<AppDb>(o => o.UseInMemoryDatabase("urunler"));
var app = builder.Build();

app.MapGet("/urunler/{id:int}", async (int id, AppDb db) =>
    await db.Urunler.FindAsync(id) is Urun urun ? Results.Ok(urun) : Results.NotFound());

app.Run();

public class Urun
{
    public int Id { get; set; }
    public string Ad { get; set; } = "";
    public decimal Fiyat { get; set; }
}

public class AppDb(DbContextOptions<AppDb> options) : DbContext(options)
{
    public DbSet<Urun> Urunler => Set<Urun>();
}

public partial class Program { }
```

**Api.Tests/UrunEndpointTests.cs**
```csharp
using System.Net;
using System.Net.Http.Json;
using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.Extensions.DependencyInjection;
using Xunit;

public class UrunEndpointTests(WebApplicationFactory<Program> factory)
    : IClassFixture<WebApplicationFactory<Program>>
{
    [Fact]
    public async Task Var_olan_urun_200_ve_urunu_doner()
    {
        using (var scope = factory.Services.CreateScope())
        {
            var db = scope.ServiceProvider.GetRequiredService<AppDb>();
            db.Urunler.Add(new Urun { Id = 1, Ad = "Kalem", Fiyat = 10m });
            await db.SaveChangesAsync();
        }

        var res = await factory.CreateClient().GetAsync("/urunler/1");

        Assert.Equal(HttpStatusCode.OK, res.StatusCode);
        var urun = await res.Content.ReadFromJsonAsync<Urun>();
        Assert.Equal("Kalem", urun!.Ad);
    }

    [Fact]
    public async Task Olmayan_urun_404_doner()
    {
        var res = await factory.CreateClient().GetAsync("/urunler/999");

        Assert.Equal(HttpStatusCode.NotFound, res.StatusCode);
    }
}
```
